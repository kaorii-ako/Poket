# The UI is one gzipped file generated by web/build.py. Regenerate it whenever
# a source file under web/ is newer, so a CSS tweak can't silently not ship.
add_custom_target(poket_web ALL
    COMMAND ${CMAKE_COMMAND} -E echo "-- checking web bundle"
    COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/../web/build.py
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/../web
    BYPRODUCTS ${CMAKE_CURRENT_SOURCE_DIR}/net/www.h
    VERBATIM
)
add_dependencies(${COMPONENT_LIB} poket_web)
